import { completeDag, key, parseCid } from "./snapshots/old/content.js";
import { encodeCar } from "./snapshots/old/car.js";
import { getBlock, getCar } from "./snapshots/new/storage.js";
import allowedCids from "./allowed-cids.json";

interface Env {
  DB: D1Database;
  BLOCKS: R2Bucket;
  BENCH_TOKEN: string;
  ROOTS: string;
}

async function authorized(request: Request, token: string): Promise<boolean> {
  const header = request.headers.get("authorization");
  if (!header || header.length > 128) return false;
  const encoder = new TextEncoder();
  const [a, b] = await Promise.all([
    crypto.subtle.digest("SHA-256", encoder.encode(header)),
    crypto.subtle.digest("SHA-256", encoder.encode(`Bearer ${token}`)),
  ]);
  const left = new Uint8Array(a);
  const right = new Uint8Array(b);
  let mismatch = 0;
  for (let index = 0; index < left.length; index++)
    mismatch |= left[index]! ^ right[index]!;
  return mismatch === 0;
}

export default {
  async fetch(request: Request, env: Env): Promise<Response> {
    if (!env.BENCH_TOKEN || env.BENCH_TOKEN.length < 32 || !env.ROOTS) {
      console.error(JSON.stringify({ event: "benchmark_configuration_missing" }));
      return Response.json({ error: "experiment unavailable" }, { status: 503 });
    }
    if (!(await authorized(request, env.BENCH_TOKEN)))
      return Response.json({ error: "unauthorized" }, { status: 401 });
    if (request.method !== "GET")
      return Response.json({ error: "read-only experiment" }, { status: 405 });
    const url = new URL(request.url);
    if (url.pathname === "/health")
      return Response.json({ experiment: "isolated-car-comparison" });
    if (url.pathname === "/inventory") {
      const listing = await env.BLOCKS.list({ limit: 1000 });
      return Response.json({
        truncated: listing.truncated,
        objects: listing.objects.map(object => ({
          key: object.key,
          size: object.size,
          etag: object.etag,
        })),
      });
    }
    const match = /^\/(legacy|indexed)\/(car|raw)\/([a-z0-9]+)$/.exec(
      url.pathname,
    );
    if (!match)
      return Response.json({ error: "not found" }, { status: 404 });
    const [, variant, format, cid] = match;
    const allowed: string[] =
      format === "car" ? JSON.parse(env.ROOTS) : allowedCids;
    if (!cid || !allowed.includes(cid))
      return Response.json({ error: "fixture not allowed" }, { status: 404 });
    const counts = { get: 0, head: 0, missing: 0, failed: 0 };
    const bucket = new Proxy(env.BLOCKS, {
      get(target, property) {
        if (property === "get")
          return async (objectKey: string, options?: R2GetOptions) => {
            counts.get++;
            try {
              const object = await target.get(objectKey, options);
              if (!object) counts.missing++;
              return object;
            } catch (error) {
              counts.failed++;
              throw error;
            }
          };
        if (property === "head")
          return async (objectKey: string) => {
            counts.head++;
            try {
              const object = await target.head(objectKey);
              if (!object) counts.missing++;
              return object;
            } catch (error) {
              counts.failed++;
              throw error;
            }
          };
        const value: unknown = Reflect.get(target, property, target);
        return typeof value === "function" ? value.bind(target) : value;
      },
    });
    try {
      let body: BodyInit;
      let size: number;
      let etag: string | null = null;
      if (variant === "legacy" && format === "car") {
        const root = parseCid(cid);
        if (!(await bucket.head(key(root))))
          return Response.json({ error: "DAG not found" }, { status: 404 });
        const dag = await completeDag(root, async link => {
          const block = await bucket.get(key(link));
          return block ? new Uint8Array(await block.arrayBuffer()) : undefined;
        });
        const bytes = await encodeCar(dag);
        body = bytes;
        size = bytes.length;
      } else if (variant === "legacy") {
        const object = await bucket.get(cid);
        if (!object)
          return Response.json({ error: "block not found" }, { status: 404 });
        body = object.body;
        size = object.size;
        etag = `"${cid}.raw"`;
      } else if (format === "car") {
        const object = await getCar({ DB: env.DB, BLOCKS: bucket }, cid);
        if (!object)
          return Response.json({ error: "DAG not found" }, { status: 404 });
        body = object.body;
        size = object.size;
        etag = `"${object.etag}"`;
      } else {
        const object = await getBlock({ DB: env.DB, BLOCKS: bucket }, cid);
        if (!object)
          return Response.json({ error: "block not found" }, { status: 404 });
        body = object.body;
        size = object.size;
        etag = `"${cid}.raw"`;
      }
      const headers = new Headers({
        "Content-Type":
          format === "car"
            ? "application/vnd.ipld.car; version=1"
            : "application/vnd.ipld.raw",
        "Content-Length": String(size),
        "Cache-Control": "no-store",
        "Referrer-Policy": "no-referrer",
        "X-Content-Type-Options": "nosniff",
        "X-Benchmark-Variant": variant!,
        "X-Benchmark-R2-Get": String(counts.get),
        "X-Benchmark-R2-Head": String(counts.head),
        "X-Benchmark-R2-Missing": String(counts.missing),
        "X-Benchmark-R2-Failed": String(counts.failed),
      });
      if (etag) headers.set("ETag", etag);
      console.log(
        JSON.stringify({
          event: "real_r2_binding_observation",
          variant,
          format,
          cid,
          ...counts,
        }),
      );
      return new Response(body, { headers });
    } catch (error) {
      console.error(
        JSON.stringify({
          event: "isolated_benchmark_failed",
          variant,
          format,
          cid,
          errorType: error instanceof Error ? error.name : "unknown",
          ...counts,
        }),
      );
      return Response.json(
        { error: "benchmark read failed", ...counts },
        { status: 503, headers: { "Cache-Control": "no-store" } },
      );
    }
  },
} satisfies ExportedHandler<Env>;
